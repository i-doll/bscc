==================
Sensor / scan loop
==================

integer scan_count = 0;

default {
    state_entry() {
        llSensorRepeat("", NULL_KEY, AGENT, 10.0, PI, 1.0);
    }

    sensor(integer detected) {
        integer i = 0;
        while (i < detected) {
            string name = llDetectedName(i);
            if (name != "") {
                llSay(0, "saw " + name);
                ++scan_count;
            }
            ++i;
        }
    }

    no_sensor() {
        llSay(0, "nobody here");
    }
}
