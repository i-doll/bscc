==================

default {
    state_entry() {
        integer n = 5;
        if (n < 0) {
            llSay(0, "negative");
        } else if (n == 0) {
            llSay(0, "zero");
        } else {
            llSay(0, "positive");
        }
    }
}

default {
    state_entry() {
        integer i;
        for (i = 0; i < 5; ++i) {
            if (i == 3) jump done;
        }
        @done;
    }
}
